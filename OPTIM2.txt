% SPRING OPTIMIZATION 
% 1. FMINCON 
% Objective function
objective = @(x) springObjective(x);
% Initial values
x0 = [0.05, 0.50, 10];
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings
options = optimoptions('fmincon','Algorithm','sqp','Display','off', ...
    'MaxFunctionEvaluations',100000);
% Run optimization
tic;
[xopt,fval] = fmincon(objective,x0,[],[],[],[],lb,ub,@springConstraints,options);
t_fmincon = toc;
% Display results
fprintf('--- fmincon ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Time = %.4f s\n\n',t_fmincon);
res.fmincon = fval;

%2. GA  (genetic algorithm)
rng(1);                                  % seed, so results can be repeated
% Objective function
objective = @(x) springObjective(x);
% Number of variables
nvars = 3;
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings
options = optimoptions('ga','Display','off');
% Run optimization 
tic;
[xopt,fval] = ga(objective,nvars,[],[],[],[],lb,ub,@springConstraints,options);
t_ga = toc;
% Display results
fprintf('--- ga ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Time = %.4f s\n\n',t_ga);
res.ga = fval;


%3. PARTICLE SWARM  (particle swarm)
rng(1);
% Objective function WITH penalty
objective = @(x) springObjective(x) + 1e5*sum(max(springConstraints(x),0));
% Number of variables
nvars = 3;
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings
options = optimoptions('particleswarm','Display','off');
% Run optimization
tic;
[xopt,~] = particleswarm(objective,nvars,lb,ub,options);
t_particleswarm = toc;
% Display results (true weight, without the penalty)
fval = springObjective(xopt);
fprintf('--- particleswarm ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Max constraint violation = %.2e\n',max([0,springConstraints(xopt)]));
fprintf('Time = %.4f s\n\n',t_particleswarm);
res.particleswarm = fval;


% 4. PATTERNSEARCH  (direct search, no gradients)
% Objective function
objective = @(x) springObjective(x);
% Initial values
x0 = [0.05, 0.50, 10];
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings
options = optimoptions('patternsearch','Display','off');
% Run optimization
tic;
[xopt,fval] = patternsearch(objective,x0,[],[],[],[],lb,ub, ...
    @springConstraints,options);
t_patternsearch = toc;
% Display results
fprintf('--- patternsearch ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Time = %.4f s\n\n',t_patternsearch);
res.patternsearch = fval;


% 5. SURROGATEOPT 
rng(1);
% Objective function returning the struct
objective = @(x) struct('Fval',springObjective(x),'Ineq',springConstraints(x));
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings (PlotFcn = [] switches off the default plot)
options = optimoptions('surrogateopt','Display','off','PlotFcn',[]);
% Run optimization
tic;
[xopt,fval] = surrogateopt(objective,lb,ub,options);
t_surrogateopt = toc;
% Display results
fprintf('--- surrogateopt ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Time = %.4f s\n\n',t_surrogateopt);
res.surrogateopt = fval;


% 6. SIMULANNEALBND  (simulated annealing)
rng(1);
% Objective function WITH penalty
objective = @(x) springObjective(x) + 1e5*sum(max(springConstraints(x),0));
% Initial values
x0 = [0.05, 0.50, 10];
% Lower and upper bounds
lb = [0.05, 0.25, 2];
ub = [2, 1.3, 15];
% Optimization settings
options = optimoptions('simulannealbnd','Display','off');
% Run optimization
tic;
[xopt,~] = simulannealbnd(objective,x0,lb,ub,options);
t_simulannealbnd = toc;
% Display results (true weight, without the penalty)
fval = springObjective(xopt);
fprintf('--- simulannealbnd ---\n');
fprintf('Optimal wire diameter = %.6f\n',xopt(1));
fprintf('Optimal coil diameter = %.6f\n',xopt(2));
fprintf('Optimal active coils = %.6f\n',xopt(3));
fprintf('Minimum spring weight objective = %.8f\n',fval);
fprintf('Max constraint violation = %.2e\n',max([0,springConstraints(xopt)]));
fprintf('Time = %.4f s\n\n',t_simulannealbnd);
res.simulannealbnd = fval;


% 7. COMPARISON GRAPH OF COMPUTATION TIME
names = {'fmincon','ga','particleswarm','patternsearch', ...
         'surrogateopt','simulannealbnd'};
times = [t_fmincon, t_ga, t_particleswarm, t_patternsearch, ...
         t_surrogateopt, t_simulannealbnd];
weights = [res.fmincon, res.ga, res.particleswarm, res.patternsearch, ...
           res.surrogateopt, res.simulannealbnd];

% Table of results
T = table(names', times', weights', ...
    'VariableNames',{'Solver','Time_s','Objective'});
disp(T);

% Bar chart of time
figure;
b = bar(times,0.6);
b.FaceColor = [0.2 0.45 0.75];
set(gca,'XTick',1:numel(names),'XTickLabel',names,'XTickLabelRotation',30);
ylabel('Computation time (s)');
title('Time taken by each solver on the spring problem');
set(gca,'YScale','log');        % log scale because the times differ a lot
grid on;
for i = 1:numel(times)
    text(i,times(i)*1.15,sprintf('%.3f s',times(i)), ...
        'HorizontalAlignment','center');
end

% FUNCTIONS (must stay at the end of the file)
function f = springObjective(x)
% Spring weight
    f = (x(3)+2)*x(1)^2*x(2);
end

function [c,ceq] = springConstraints(x)
% Constraint function (same as your original)

x1 = x(1);
x2 = x(2);
x3 = x(3);

c(1) = 1-(x2^3*x3)/(71785*x1^4);

c(2) = (4*x2^2-x1*x2)/(12566*x1^3*(x2-x1)) ...
       +1/(5108*x1^2)-1;

c(3) = 1-(140.45*x1)/(x2^2*x3);

c(4) = (x1+x2)/1.5-1;

ceq = [];

c(~isfinite(c)) = 1e6;

end
