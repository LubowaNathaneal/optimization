% ORE BLENDING 
%A mining company operates two ore sources. Source A contians high grade
% ore with with copper content of 2.5 percent while Source B contains low
% grade ore with a copper content of 0.8 percent. the company wants to
% blend the two ore sources to maximize its revenue contribution. the
% contribution from A is estimated at $45 per tonne while that from source
% B is $25 per tonne. the processing pant requires a minimum of 100 tonnes
% and can process a maximum of 1200 tonnes per day. the company can supply
% at most 500 tonnes from A and 900 tonnes from B. the final blended ore
% must contain at least 1.5 percent coppper. determine quantites of ore to
% obtain from each source to maximize the total contribution while
% satisfying all the constraints.

% Variables: x(1) = tonnes from Source A (2.5% Cu, 
%            x(2) = tonnes from Source B (0.8% Cu, 
% Maximize 45*xA + 25*xB   ->  solvers minimize, so the objective is negated
% Constraints:  1000 <= xA + xB <= 1200     (plant minimum / maximum)
%               0 <= xA <= 500,  0 <= xB <= 900   (supply limits)
%               (2.5xA + 0.8xB)/(xA + xB) >= 1.5  ->  -xA + 0.7xB <= 0  

clear; clc; close all;

% 1. FMINCON  
% Problem data
x0 = [450, 550];                % starting point 
lb = [0, 0];                    % lower bounds
ub = [500, 900];                % upper bounds = supply limits
% Linear inequality constraints  A*x <= b
A = [ 1    1;                   %  xA + xB <= 1200
     -1   -1;                   %  xA + xB >= 1000
     -1  0.7];                  %  copper grade >= 1.5%
b = [1200; -1000; 0];
Aeq = []; beq = [];             % no equality constraints
fun = @(x) -(45*x(1) + 25*x(2));
objective = fun;
nvars = 2;
% Optimization settings
options = optimoptions('fmincon','Display','off','Algorithm','sqp');
% Run optimization
tic;
[xopt,fval] = fmincon(objective,x0,A,b,Aeq,beq,lb,ub,[],options);
t_fmincon = toc;
% Display results
fprintf('--- fmincon ---\n');
fprintf('Ore from Source A = %.4f t\n',xopt(1));
fprintf('Ore from Source B = %.4f t\n',xopt(2));
fprintf('Total ore = %.4f t\n',sum(xopt));
fprintf('Copper grade = %.4f %%\n',100*(2.5*xopt(1)+0.8*xopt(2))/sum(xopt));
fprintf('Maximum contribution = $%.2f\n',-fval);
fprintf('Time = %.4f s\n\n',t_fmincon);
res.fmincon = -fval;

% 3. PARTICLESWARM  
rng(1);                                  
% Problem data
x0 = [450, 550];                % starting point 
lb = [0, 0];                    % lower bounds
ub = [500, 900];                % upper bounds = supply limits
% Linear inequality constraints  A*x <= b
A = [ 1    1;                   %  xA + xB <= 1200
     -1   -1;                   %  xA + xB >= 1000
     -1  0.7];                  %  copper grade >= 1.5%
b = [1200; -1000; 0];
Aeq = []; beq = [];             % no equality constraints
fun = @(x) -(45*x(1) + 25*x(2));
% Constraints are added to the objective as a penalty (weight 1e3)
objective = @(x) fun(x) + 1e3*(sum(max(A*x(:)-b,0)));
nvars = 2;
% Optimization settings
options = optimoptions('particleswarm','Display','off');
% Run optimization
tic;
[xopt,fval] = particleswarm(objective,nvars,lb,ub,options);
t_particleswarm = toc;
% Display results
fval = fun(xopt);          % true objective, without the penalty
fprintf('--- particleswarm ---\n');
fprintf('Ore from Source A = %.4f t\n',xopt(1));
fprintf('Ore from Source B = %.4f t\n',xopt(2));
fprintf('Total ore = %.4f t\n',sum(xopt));
fprintf('Copper grade = %.4f %%\n',100*(2.5*xopt(1)+0.8*xopt(2))/sum(xopt));
fprintf('Maximum contribution = $%.2f\n',-fval);
fprintf('Max constraint violation = %.2e\n',max([0; A*xopt(:)-b]));
fprintf('Time = %.4f s\n\n',t_particleswarm);
res.particleswarm = -fval;
